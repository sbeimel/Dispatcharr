```python
def sanitize_proxy_url(proxy_url):
    """Remove credentials from proxy URL for safe logging.
    
    Args:
        proxy_url: Proxy URL that may contain username:password
        
    Returns:
        Sanitized URL with credentials replaced by ***
        
    Example:
        sanitize_proxy_url("http://user:pass@proxy:8080")
        Returns: "http://proxy:8080"
    """
    if not proxy_url:
        return proxy_url
    try:
        from urllib.parse import urlparse, urlunparse
        parsed = urlparse(proxy_url)
        if parsed.username or parsed.password:
            netloc = f"{parsed.hostname}"
            if parsed.port:
                netloc += f":{parsed.port}"
            sanitized = parsed._replace(netloc=netloc)
            return urlunparse(sanitized)
    except Exception:
        return "[invalid proxy URL]"
    return proxy_url

def validate_proxy_url(proxy_url):
    """Validate proxy URL format.
    
    Args:
        proxy_url: Proxy URL string to validate
        
    Raises:
        ValidationError: If URL format is invalid or uses unsupported protocol
    """
    from django.core.exceptions import ValidationError
    from urllib.parse import urlparse
    
    try:
        parsed = urlparse(proxy_url)
        if parsed.scheme not in ('http', 'https', 'socks5'):
            raise ValidationError(
                f"Proxy URL must use http://, https://, or socks5:// protocol, got: {parsed.scheme or '(none)'}"
            )
        if not parsed.netloc:
            raise ValidationError(
                "Proxy URL must include host and optional port (e.g., http://proxy.example.com:8080)"
            )
    except ValidationError:
        raise
    except Exception as e:
        raise ValidationError(f"Invalid proxy URL format: {e}")
```

```python
def build_command(self, stream_url, user_agent, channel_id=None, proxy=None):
    
    if self.is_proxy():
        return []
    
    replacements = {
        "{streamUrl}": stream_url,
        "{userAgent}": user_agent,
        "{channelId}": str(channel_id) if channel_id else "",
        "{proxy}": proxy or "",
    }
    
    # Split the command and iterate through each part to apply replacements
    cmd = [self.command] + [
        self._replace_in_part(part, replacements)
        for part in shlex_split(self.parameters)
    ]
    
    # Automatic ffmpeg -http_proxy injection when proxy is configured and no {proxy} placeholder
    if proxy and self.command.lower() == 'ffmpeg' and '{proxy}' not in self.parameters:
        # Try to insert -http_proxy before -i flag
        try:
            i_index = cmd.index('-i')
            cmd.insert(i_index, proxy)
            cmd.insert(i_index, '-http_proxy')
        except ValueError:
            # No -i flag found - append -http_proxy at end
            logger.warning(f"FFmpeg command has no -i flag, appending -http_proxy at end")
            cmd.extend(['-http_proxy', proxy])
    
    return cmd
```

```python
    from apps.proxy.config import STREAM_COOLDOWN_ENABLED, STREAM_COOLDOWN_MINUTES
    from apps.proxy import redis_keys
    
    if not STREAM_COOLDOWN_ENABLED:
        return
    
    try:
        if hasattr(self, 'current_stream_id') and self.current_stream_id:
            from apps.channels.models import Stream
            stream = Stream.objects.get(id=self.current_stream_id)
            stream_profile = stream.get_stream_profile()
            
            cooldown_key = redis_keys.stream_cooldown(stream.id, stream_profile.id)
            cooldown_seconds = STREAM_COOLDOWN_MINUTES * 60
            
            if hasattr(self.buffer, 'redis_client') and self.buffer.redis_client:
                self.buffer.redis_client.setex(
                    cooldown_key,
                    cooldown_seconds,
                    "1"
                )
                logger.info(
                    f"Stream {stream.id} with profile {stream_profile.name} "
                    f"in cooldown for {STREAM_COOLDOWN_MINUTES} minutes"
                )
    except Exception as e:
        logger.debug(f"Could not set stream cooldown: {e}")
```

```python
def validate_and_normalize_uuid(uuid_string):
    """Validate and normalize UUID string.
    
    Args:
        uuid_string: UUID string to validate (with or without hyphens)
        
    Returns:
        Normalized UUID string with hyphens
        
    Raises:
        ValueError: If UUID is invalid
    """
    import uuid
    
    if not uuid_string:
        raise ValueError("UUID string cannot be empty")
    
    # Remove hyphens for validation
    uuid_clean = uuid_string.replace('-', '')
    
    # Validate length
    if len(uuid_clean) != 32:
        raise ValueError(f"Invalid UUID length: {len(uuid_clean)} (expected 32 hex digits)")
    
    # Validate hex digits
    try:
        int(uuid_clean, 16)
    except ValueError:
        raise ValueError(f"Invalid UUID format: contains non-hex characters")
    
    # Convert to UUID object to validate structure
    try:
        uuid_obj = uuid.UUID(uuid_clean)
        return str(uuid_obj)  # Returns normalized format with hyphens
    except Exception as e:
        raise ValueError(f"Invalid UUID: {e}")
```

#### Feature 3.3: Adaptive Health Monitor
**File:** `Dispatcharr-0.30.0/apps/proxy/live_proxy/input/manager.py`

**Add adaptive thresholds based on stream history:**
```python
def _calculate_adaptive_threshold(self, metric_name):
    """Calculate adaptive threshold based on stream history.
    
    Args:
        metric_name: Name of metric (e.g., 'chunk_interval', 'bitrate')
        
    Returns:
        Adaptive threshold value
    """
    # Get historical data from Redis
    history_key = f"stream_history:{self.channel_id}:{metric_name}"
    
    if not hasattr(self.buffer, 'redis_client') or not self.buffer.redis_client:
        return self._get_default_threshold(metric_name)
    
    try:
        # Get last N values
        history = self.buffer.redis_client.lrange(history_key, 0, 99)
        if not history:
            return self._get_default_threshold(metric_name)
        
        # Calculate median and standard deviation
        values = [float(v) for v in history]
        median = sorted(values)[len(values) // 2]
        stddev = (sum((v - median) ** 2 for v in values) / len(values)) ** 0.5
        
        # Adaptive threshold: median + 2*stddev
        adaptive = median + (2 * stddev)
        return adaptive
        
    except Exception as e:
        logger.debug(f"Could not calculate adaptive threshold: {e}")
        return self._get_default_threshold(metric_name)

def _get_default_threshold(self, metric_name):
    """Get default threshold values."""
    defaults = {
        'chunk_interval': 5.0,  # seconds
        'bitrate': 100000,  # bytes/sec
    }
    return defaults.get(metric_name, 0)
```

```python
            # Return URL that includes failover logic

        return primary_url
        
    except Exception as e:
```
